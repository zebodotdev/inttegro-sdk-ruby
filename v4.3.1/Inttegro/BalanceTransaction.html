def source_id
  payment = payment_id
  refund = refund_id
  if type == Inttegro::BalanceTransactionType::PAYMENT && payment && !payment.empty? && refund.nil?
    return payment
  end
  if type == Inttegro::BalanceTransactionType::REFUND && refund && !refund.empty? && payment.nil?
    return refund
  end

  nil
end
